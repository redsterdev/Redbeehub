local Rayfield = loadstring(game:HttpGet("https://sirius.menu/rayfield"))()
local Players = game:GetService("Players")
local UIS = game:GetService("UserInputService")
local RS = game:GetService("RunService")
local REP = game:GetService("ReplicatedStorage")
local lp = Players.LocalPlayer

local cfg = {reach=6, enabled=true, priority=false, aimbot=false, autogk=false}
local ball, hrp, torso = nil, nil, nil
local goalA = REP.FF4v4.Pitch.GoalA
local goalB = REP.FF4v4.Pitch.GoalB
local priConn, touchConn, gkConn = nil, nil, nil
local highlightA, highlightB = nil, nil
local lastShot = 0
local GRAVITY = workspace.Gravity
local gkLastPos, gkVel = {}, {}
local ballLastPos, ballVelEst = nil, Vector3.zero

-- Find Ball
local function findBall()
    local ok, val = pcall(function() return REP.GoalLocal.BallPosPostGoal.Value end)
    if ok and val and val:IsA("BasePart") then return val end
    for _, d in ipairs(workspace:GetDescendants()) do
        if d.Name == "SoccerBall" and d:IsA("BasePart") then return d end
    end
    return nil
end

-- Goal functions
local function goalPos(g)
    if not g then return Vector3.zero end
    if g:IsA("Model") then
        local ok, cf = pcall(function() return g:GetPivot() end)
        if ok then return cf.Position end
        local sum, n = Vector3.zero, 0
        for _, p in ipairs(g:GetDescendants()) do
            if p:IsA("BasePart") then sum = sum + p.Position n = n + 1 end
        end
        return n > 0 and sum / n or Vector3.zero
    end
    return g.Position
end

local function goalCF(g)
    if not g then return CFrame.new() end
    if g:IsA("Model") then
        local ok, cf = pcall(function() return g:GetPivot() end)
        if ok then return cf end
    end
    return g.CFrame
end

local function goalSz(g)
    if not g then return Vector3.new(12, 4, 2) end
    if g:IsA("Model") then
        local ok, _, sz = pcall(function() return g:GetBoundingBox() end)
        if ok and sz then return sz end
    end
    return g.Size
end

local function getEnemyGoal()
    if not hrp then return goalB end
    return (goalPos(goalA) - hrp.Position).Magnitude > (goalPos(goalB) - hrp.Position).Magnitude and goalA or goalB
end

local function getOwnGoal()
    return getEnemyGoal() == goalA and goalB or goalA
end

-- Highlights
local function updateHighlights()
    if highlightA then highlightA:Destroy() highlightA = nil end
    if highlightB then highlightB:Destroy() highlightB = nil end
    if not cfg.aimbot or not goalA or not goalB then return end
    local enemy = getEnemyGoal()
    local function hl(g, col)
        local h = Instance.new("Highlight")
        h.FillColor = col
        h.OutlineColor = col
        h.FillTransparency = 0.4
        h.OutlineTransparency = 0
        h.Adornee = g
        h.Parent = workspace
        return h
    end
    highlightA = hl(goalA, goalA == enemy and Color3.fromRGB(255, 40, 40) or Color3.fromRGB(40, 100, 255))
    highlightB = hl(goalB, goalB == enemy and Color3.fromRGB(255, 40, 40) or Color3.fromRGB(40, 100, 255))
end

-- Track GK
RS.Heartbeat:Connect(function()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp and p.Character then
            local root = p.Character:FindFirstChild("HumanoidRootPart")
            if root then
                local id = p.UserId
                if gkLastPos[id] then gkVel[id] = (root.Position - gkLastPos[id]) / 0.016 end
                gkLastPos[id] = root.Position
            end
        end
    end
end)

local function findGK(enemyGoal)
    local closest, bestDist, closestId = nil, 35, nil
    local ep = goalPos(enemyGoal)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp and p.Character then
            local root = p.Character:FindFirstChild("HumanoidRootPart")
            if root then
                local d = (root.Position - ep).Magnitude
                if d < bestDist then
                    bestDist = d
                    closest = root
                    closestId = p.UserId
                end
            end
        end
    end
    return closest, closestId
end

-- Shot solving
local function solveShot(origin, target, spd)
    local flat = Vector3.new(target.X - origin.X, 0, target.Z - origin.Z)
    local dx = flat.Magnitude
    local dy = target.Y - origin.Y
    if dx < 0.01 then return Vector3.new(0, spd, 0) end
    local t = dx / spd
    local vy = (dy / t) + (0.5 * GRAVITY * t)
    return Vector3.new(flat.Unit.X * spd, vy, flat.Unit.Z * spd)
end

-- Get best target
local function getBestTarget(enemyGoal, gkRoot, gkId)
    local cf = goalCF(enemyGoal)
    local gp = cf.Position
    local right = cf.RightVector
    local fwd = cf.LookVector
    local width = goalSz(enemyGoal).X
    local zones = {}
    for _, xOff in ipairs({-0.32, 0, 0.32}) do
        for _, yOff in ipairs({1.0, 2.5}) do
            table.insert(zones, gp + right * (width * xOff) + Vector3.new(0, yOff, 0) + fwd * 1.5)
        end
    end
    if not gkRoot then
        local side = math.random() > 0.5 and 0.32 or -0.32
        return gp + right * (width * side) + Vector3.new(0, 1.0, 0) + fwd * 1.5
    end
    local dist = ball and (gp - ball.Position).Magnitude or 20
    local gkPred = gkRoot.Position + (gkVel[gkId] or Vector3.zero) * (dist / 55)
    local best, bestScore = zones[1], 0
    for _, z in ipairs(zones) do
        local score = (z - gkPred).Magnitude
        if z.Y < gp.Y + 1.5 then score = score * 1.4 end
        if score > bestScore then bestScore = score best = z end
    end
    return best
end

-- Shoot
local function shoot(target)
    if not ball or not ball.Parent then return end
    local origin = ball.Position
    for _, v in ipairs(ball:GetChildren()) do
        if v:IsA("BodyVelocity") or v:IsA("BodyForce") or v:IsA("BodyMover") then v:Destroy() end
    end
    local vel
    for _, spd in ipairs({45, 55, 65, 75}) do
        local v = solveShot(origin, target, spd)
        if math.abs(v.Y) < spd * 2.5 then vel = v break end
    end
    if not vel then vel = (target - origin).Unit * 55 + Vector3.new(0, 8, 0) end
    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(1e6, 1e6, 1e6)
    bv.P = 1e6
    bv.Velocity = vel
    bv.Parent = ball
    task.delay(0.07, function() if bv and bv.Parent then bv:Destroy() end end)
end

-- Aimbot
local function startTouch()
    if touchConn then touchConn:Disconnect() touchConn = nil end
    if not cfg.aimbot then return end
    touchConn = RS.Heartbeat:Connect(function()
        if not cfg.enabled or not ball or not ball.Parent or not hrp then return end
        if tick() - lastShot < 0.45 then return end
        if (ball.Position - hrp.Position).Magnitude > math.min(cfg.reach, 4.5) then return end
        lastShot = tick()
        local enemy = getEnemyGoal()
        if not enemy then return end
        local gkRoot, gkId = findGK(enemy)
        shoot(getBestTarget(enemy, gkRoot, gkId))
    end)
end

-- Ball prediction
local function predictBallIntercept(bPos, bVel, targetY)
    local a = 0.5 * GRAVITY
    local b2 = -bVel.Y
    local c = targetY - bPos.Y
    local disc = b2 * b2 - 4 * a * c
    if disc < 0 then return nil end
    local t1 = (-b2 - math.sqrt(disc)) / (2 * a)
    local t2 = (-b2 + math.sqrt(disc)) / (2 * a)
    local t = nil
    if t1 > 0.05 then t = t1 end
    if t2 > 0.05 and (not t or t2 < t) then t = t2 end
    if not t then return nil end
    return Vector3.new(bPos.X + bVel.X * t, targetY, bPos.Z + bVel.Z * t)
end

-- Auto GK
local function startAutoGK()
    if gkConn then gkConn:Disconnect() gkConn = nil end
    if not cfg.autogk then return end
    gkConn = RS.Heartbeat:Connect(function(dt)
        if not ball or not ball.Parent or not hrp then return end
        local bPos = ball.Position
        if ballLastPos then ballVelEst = (bPos - ballLastPos) / dt end
        ballLastPos = bPos
        local own = getOwnGoal()
        if not own then return end
        local gp = goalPos(own)
        local cf = goalCF(own)
        local toBall = bPos - gp
        if toBall:Dot(ballVelEst) < 0 and (bPos - gp).Magnitude > 35 then return end
        local intercept = predictBallIntercept(bPos, ballVelEst, hrp.Position.Y)
        if not intercept then intercept = Vector3.new(bPos.X, hrp.Position.Y, bPos.Z) end
        local width = goalSz(own).X * 0.45
        local right = cf.RightVector
        local proj = math.clamp((intercept - gp):Dot(right), -width, width)
        local clamped = Vector3.new((gp + right * proj).X, hrp.Position.Y, (gp + right * proj).Z)
        local urgency = math.clamp(ballVelEst.Magnitude / 20, 0.1, 1.0)
        hrp.CFrame = CFrame.new(hrp.Position:Lerp(clamped, math.clamp(urgency * 0.35, 0.08, 0.35))) * (hrp.CFrame - hrp.CFrame.Position)
    end)
end

-- Setup
local function setup(char)
    task.wait(0.5)
    hrp = char:FindFirstChild("HumanoidRootPart")
    torso = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
    ball = findBall()
    ballLastPos = nil
    ballVelEst = Vector3.zero
    if cfg.aimbot then startTouch() end
    if cfg.autogk then startAutoGK() end
    updateHighlights()
end

if lp.Character then setup(lp.Character) end
lp.CharacterAdded:Connect(function(c)
    if priConn then priConn:Disconnect() priConn = nil end
    if touchConn then touchConn:Disconnect() touchConn = nil end
    if gkConn then gkConn:Disconnect() gkConn = nil end
    setup(c)
end)
